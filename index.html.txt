<script>
function buscar() {
    const cedula = document.getElementById("cedula").value;
    const resultado = document.getElementById("resultado");

    // 🔥 BASE DE DATOS LOCAL (puedes agregar más personas aquí)
    const datos = {
        "1019603560": {
            nombre: "Manuel Suarez",
            certificados: [
                {
                    nombre: "Certificado EPS",
                    archivo: "certificados/1019603560_eps.pdf"
                }
            ]
        }
    };

    if (datos[cedula]) {
        const persona = datos[cedula];

        let html = `
            <h4>${persona.nombre}</h4>
            <ul class="list-group mt-3">
        `;

        persona.certificados.forEach(cert => {
            html += `
                <li class="list-group-item d-flex justify-content-between align-items-center">
                    ${cert.nombre}
                    <a href="${cert.archivo}" target="_blank" class="btn btn-sm btn-success">
                        Ver PDF
                    </a>
                </li>
            `;
        });

        html += "</ul>";

        resultado.innerHTML = html;

    } else {
        resultado.innerHTML = `
            <div class="alert alert-danger mt-3">
                No se encontró la cédula
            </div>
        `;
    }
}
</script>
